cmake_minimum_required(VERSION 3.18)
project(motion_account_vault LANGUAGES CXX)
add_executable(motion-account-vault main.cpp)
target_compile_features(motion-account-vault PRIVATE cxx_std_17)
if(APPLE)
  target_link_libraries(motion-account-vault PRIVATE "-framework Security" "-framework CoreFoundation")
elseif(WIN32)
  target_link_libraries(motion-account-vault PRIVATE Advapi32)
else()
  message(FATAL_ERROR "Account vault supports macOS and Windows only")
endif()
