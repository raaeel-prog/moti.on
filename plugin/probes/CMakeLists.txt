target_link_libraries(MotionGestureProbe PRIVATE "-framework Foundation")
set_target_properties(MotionGestureProbe PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION plugin PREFIX "" CXX_VISIBILITY_PRESET hidden MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist")
